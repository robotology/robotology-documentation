43 std::size_t cornersCount() const

56
57 std::vector<cv::Point3f> createObjectPoints() const
58 {
59 std::vector<cv::Point3f> points;
60
61 if(!isValid())
62 {
63 return points;
64 }
65
66 points.reserve(cornersCount());
67
68 for (int r = 0; r < cornersY; ++r)
69 {
70 for(int c = 0; c < cornersX; ++c)
71 {
72 points.emplace_back(
73 static_cast<float>(c * squareSizeMeters),
74 static_cast<float>(r * squareSizeMeters),
75 0.0F
76 );
77 }
78 }
79
80 return points;
81 }
