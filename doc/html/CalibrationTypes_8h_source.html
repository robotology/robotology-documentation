209 struct CameraCalibrationResult
210 {
211 CameraModel model{CameraModel::Pinhole};
212 cv::Size imageSize;
213
214 // 3x3 intrinsic camera matrix
215 cv::Mat K;
216
217 // Pinhole distortion coefficients: [k1, k2, p1, p2]
218 // Fisheye distortion coefficients: [k1, k2, k3, k4]
219 cv::Mat D;
220
221 // Board pose for each accepted observation
222 std::vector<cv::Mat> rotationVectors;
223 std::vector<cv::Mat> translationVectors;
224
225 // Reprojection RMS for every accepted calibration observation.
226 std::vector<double> perViewRms;
227
228 double rms{-1.0};
229
230 bool isValid() const
231 {
232 if(model != CameraModel::Pinhole &&
233 model != CameraModel::Fisheye)
234 return false;
235
236 if(imageSize.width <=0 || imageSize.height <= 0)
237 return false;
238
239 if(K.empty() ||
240 K.rows != 3 ||
241 K.cols != 3 ||
242 K.type() != CV_64F ||
243 !cv::checkRange(K, true))
244 return false;
245
246 if(D.empty() ||
247 D.total() != 4 ||
248 D.type() != CV_64F ||
249 !cv::checkRange(D, true))
250 return false;
251
252 if(K.at<double>(0, 0) <= 0.0 ||
253 K.at<double>(1,1) <= 0.0)
254 return false;
255
256 if(!std::isfinite(rms) || rms < 0.0)
257 return false;
258
259 if(rotationVectors.size() != translationVectors.size() ||
260 rotationVectors.size() != perViewRms.size())
261 return false;
262
263 for (size_t i = 0; i < rotationVectors.size(); i++)
264 {
265 const cv::Mat& rotation = rotationVectors[i];
266 const cv::Mat& translation = translationVectors[i];
267
268 if(rotation.empty() ||
269 rotation.total() != 3 ||
270 rotation.type() != CV_64F ||
271 !cv::checkRange(rotation, true))
272 return false;
273
274 if(translation.empty() ||
275 translation.total() != 3 ||
276 translation.type() != CV_64F ||
277 !cv::checkRange(translation, true))
278 return false;
279
280 if(!std::isfinite(perViewRms[i]) ||
281 perViewRms[i] < 0.0)
282 return false;
283 }
284
285 return true;
286 }

289 struct StereoCalibrationResult
290 {
291 // Transform convention:
292 //
293 // X_right = R * X_left + T
294 //
295 // R: 3x3 rotation matrix
296 // T: 3x1 translation vector.
297 cv::Mat R;
298 cv::Mat T;
299
300 double rms{-1.0};
301
302 bool isValid() const
303 {
304 if(R.empty() ||
305 R.rows != 3 ||
306 R.cols != 3 ||
307 R.type() != CV_64F ||
308 !cv::checkRange(R, true)
309 )
310 return false;
311
312 if(T.empty() ||
313 T.total() != 3 ||
314 T.type() != CV_64F ||
315 !cv::checkRange(T, true))
316 return false;
317
318 if(!std::isfinite(rms) ||
319 rms < 0.0 )
320 return false;
321
322 const cv::Mat identity = cv::Mat::eye(3,3,CV_64F);
323 if(std::abs(cv::determinant(R) - 1.0) >= 1e-3 ||
324 cv::norm(R.t() * R - identity, cv::NORM_INF) >= 1e-3 ||
325 cv::norm(T) <= 1e-9)
326 return false;
327
328 return true;
329 }
330 };
