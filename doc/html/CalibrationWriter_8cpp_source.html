101{
102 if (!observation.isValid() || !isFinite(observation.leftTimestampSeconds) ||
103 !isFinite(observation.rightTimestampSeconds) ||
104 !isFinite(observation.timestampDeltaSeconds))

112 const cv::Point2f& rightPoint = observation.rightImagePoints[index];
113 if (!isFinite(objectPoint.x) || !isFinite(objectPoint.y) || !isFinite(objectPoint.z) ||

128bool CalibrationWriter::write(const std::string& outputFile,
129 const CalibrationResult& result,
130 std::string& errorMessage) const
131{
132 errorMessage.clear();
133 if (outputFile.empty())
134 {
135 errorMessage = "Calibration output filename is empty.";
136 return false;
137 }
138 if (!validateResult(result, errorMessage))
139 {
140 return false;
141 }
142
143 const std::string temporaryFile = outputFile + ".tmp";
144 std::ofstream output(temporaryFile.c_str(), std::ios::out | std::ios::trunc);
145 if (!output.is_open())
146 {
147 errorMessage = "Could not open temporary calibration file '" + temporaryFile +
148 "': " + std::strerror(errno);
149 return false;
150 }
151
152 output << std::setprecision(17);
153 output << "# Generated by stereoCalib. Do not edit while calibration is running.\n";
154 switch (result.mode)
155 {
156 case CalibrationMode::MonocularLeft: output << "calibrationMode MonocularLeft\n\n"; break;
157 case CalibrationMode::MonocularRight: output << "calibrationMode MonocularRight\n\n"; break;
158 case CalibrationMode::MonocularBoth: output << "calibrationMode MonocularBoth\n\n"; break;
159 case CalibrationMode::StereoFull: output << "calibrationMode StereoFull\n\n"; break;
160 }
161
162 const bool writesLeftCamera = result.mode != CalibrationMode::MonocularRight;
163 const bool writesRightCamera = result.mode != CalibrationMode::MonocularLeft;
164 const std::string projection = (result.model == CameraModel::Pinhole) ? "pinhole" : "fisheye";
165 if (writesLeftCamera)
166 {
167 writeCamera(output, "LEFT", projection, result.leftCamera);
168 }
169 if (writesRightCamera)
170 {
171 writeCamera(output, "RIGHT", projection, result.rightCamera);
172 }
173
174 if (result.mode == CalibrationMode::StereoFull)
175 {
176 cv::Mat homogeneousTransform = cv::Mat::eye(4, 4, CV_64F);
177 result.stereo.R.copyTo(homogeneousTransform(cv::Rect(0, 0, 3, 3)));
178 result.stereo.T.reshape(1, 3).copyTo(homogeneousTransform(cv::Rect(3, 0, 1, 3)));
179
180 output << "[STEREO_DISPARITY]\n";
181 // This is deliberately left-to-right: Xright = R * Xleft + T.
182 writeMatrix(output, "HN", homogeneousTransform);
183 writeMatrix(output, "R", result.stereo.R);
184 writeMatrix(output, "T", result.stereo.T);
185 }
186
187 output << "[CALIBRATION_QUALITY]\n";
188 if (writesLeftCamera)
189 {
190 output << "leftRms " << result.leftCamera.rms << "\n";
191 }
192 if (writesRightCamera)
193 {
194 output << "rightRms " << result.rightCamera.rms << "\n";
195 }
196 if (result.mode == CalibrationMode::StereoFull)
197 {
198 output << "stereoRms " << result.stereo.rms << "\n";
199 output << "baseline " << result.quality.baseline << "\n";
200 output << "synchronizedPairs " << result.quality.synchronizedPairs << "\n";
201 output << "acceptedObservations " << result.quality.acceptedObservations << "\n";
202 output << "rejectedDetections " << result.quality.rejectedDetections << "\n";
203 output << "meanTimestampDeltaMs " << result.quality.meanTimestampDeltaMs << "\n";
204 output << "maxTimestampDeltaMs " << result.quality.maxTimestampDeltaMs << "\n";
205 }
206
207 output.flush();
208 if (!output.good())
209 {
210 errorMessage = "Failed while writing temporary calibration file '" + temporaryFile + "'.";
211 return false;
212 }
213 output.close();
214 if (output.fail())
215 {
216 errorMessage = "Could not close temporary calibration file '" + temporaryFile + "'.";
217 return false;
218 }
219
220 std::ifstream validation(temporaryFile.c_str());
221 if (!validation.good())
222 {
223 errorMessage = "Could not validate temporary calibration file '" + temporaryFile + "'.";
224 return false;
225 }
226 validation.close();
227 if (std::rename(temporaryFile.c_str(), outputFile.c_str()) != 0)
228 {
229 errorMessage = "Could not replace calibration file '" + outputFile + "' with temporary file: " +
230 std::strerror(errno);
231 return false;
232 }
233 return true;

238 std::string& errorMessage) const
239{
240 errorMessage.clear();

245 }

256 try
257 {

262 return false;
263 }
264
265 storage << "observations" << "[";
266 for (std::size_t index = 0; index < observations.size(); ++index)
267 {
268 const StereoObservation& observation = observations[index];
269 storage << "{";
270 storage << "index" << static_cast<int>(index);
271 storage << "imageWidth" << observation.imageSize.width;
272 storage << "imageHeight" << observation.imageSize.height;
273 storage << "leftTimestampSeconds" << observation.leftTimestampSeconds;
274 storage << "rightTimestampSeconds" << observation.rightTimestampSeconds;
275 storage << "timestampDeltaSeconds" << observation.timestampDeltaSeconds;
276 storage << "leftSequenceNumber" << std::to_string(observation.leftSequenceNumber);
277 storage << "rightSequenceNumber" << std::to_string(observation.rightSequenceNumber);
278 storage << "leftImageFilename" << observation.leftImageFilename;
279 storage << "rightImageFilename" << observation.rightImageFilename;
280 storage << "objectPoints" << cv::Mat(observation.objectPoints);
281 storage << "leftImagePoints" << cv::Mat(observation.leftImagePoints);
282 storage << "rightImagePoints" << cv::Mat(observation.rightImagePoints);
283 storage << "}";
284 }
285 storage << "]";
286 storage.release();
287
288 cv::FileStorage validation(temporaryFile, cv::FileStorage::READ);

294 return false;
295 }
296 }
297 catch (const cv::Exception& exception)

300 return false;
301 }

303 {
304 errorMessage = "Could not replace observation file '" + outputFile + "' with temporary file: " +
305 std::strerror(errno);
306 return false;
307 }
