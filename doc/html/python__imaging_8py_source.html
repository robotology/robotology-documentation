56class SobelFilter:
57 """Object that reads an image from a port, transforms it and sends the result to another port.
58
59 The object reads RGB images and sends monochrome ones with calculated sobel filter.
60 Note: this class uses the alternative implementation of setExternal together with numpy array interface.
61 """
62
63 def __init__(self, input_port_name, output_port_name):
64 # Prepare ports
65 self._input_port = yarp.Port()
66 self._input_port_name = input_port_name
67 self._input_port.open(self._input_port_name)
68
69 self._output_port = yarp.Port()
70 self._output_port_name = output_port_name
71 self._output_port.open(self._output_port_name)
72
73 # Prepare image buffers
74 # Input
75 self._input_buf_image = yarp.ImageRgb()
76 self._input_buf_image.resize(320, 240)
77 self._input_buf_array = numpy.zeros((240, 320, 3), dtype = numpy.uint8)
78 self._input_buf_image.setExternal(self._input_buf_array.__array_interface__['data'][0], self._input_buf_array.shape[1], self._input_buf_array.shape[0])
79 # Output
80 self._output_buf_image = yarp.ImageFloat()
81 self._output_buf_image.resize(320, 240)
82 self._output_buf_array = numpy.zeros((240, 320), dtype = numpy.float32)
83 self._output_buf_image.setExternal(self._output_buf_array.__array_interface__['data'][0], self._output_buf_array.shape[1], self._output_buf_array.shape[0])
84
85 def run(self):
86 while True: # Break this by hitting CTRL-C
87 # Read an image from the port
88 self._input_port.read(self._input_buf_image)
89
90 # Make sure the image has not been re-allocated
91 assert self._input_buf_array.__array_interface__['data'][0] == self._input_buf_image.getRawImage().__long__()
92
93 # Let's calculate the sobel filter
94 self._output_buf_array[:,:] = scipy.ndimage.filters.sobel(self._input_buf_array.mean(2) / 255.)
95 # Normalise the result
96 self._output_buf_array -= self._output_buf_array.min()
97 self._output_buf_array /= self._output_buf_array.max()
98 self._output_buf_array *= 255.
99
100 # Send the result to the output port
101 self._output_port.write(self._output_buf_image)
102
103 def cleanup(self):
104 self._input_port.close()
105 self._output_port.close()
