61bool Node::removeNeighbor(const string &name)
62{
63 map<string,Node*>::iterator it=neighbors.find(name);
64 if (it!=neighbors.end())
65 {
66 neighbors.erase(it);
67 return true;
68 }
69 else
70 return false;
