183void ObjectsManager::manage(yarp::os::Bottle *msg)
184{
185 std::string cmd=msg->get(0).asString();
186
187 if (cmd=="reset")
188 {
189 for (int i=0; i<(int)mObjectsRoot.size(); ++i) delete mObjectsRoot[i];
190 mObjectsRoot.clear();
191
192 for (int i=0; i<(int)mObjectsWorld.size(); ++i) delete mObjectsWorld[i];
193 mObjectsWorld.clear();
194
195 for (int i=0; i<(int)mTrajectoriesRoot.size(); ++i) delete mTrajectoriesRoot[i];
196 mTrajectoriesRoot.clear();
197
198 for (int i=0; i<(int)mTrajectoriesWorld.size(); ++i) delete mTrajectoriesWorld[i];
199 mTrajectoriesWorld.clear();
200
201 return;
202 }
203
204 if (cmd=="delete")
205 {
206 std::string name(msg->get(1).asString().c_str());
207
208 int size=(int)mObjectsRoot.size();
209 for (int i=0; i<size; ++i)
210 {
211 if (mObjectsRoot[i] && *mObjectsRoot[i]==name)
212 {
213 --size;
214 delete mObjectsRoot[i];
215 for (int j=i; j<size; ++j) mObjectsRoot[j]=mObjectsRoot[j+1];
216 mObjectsRoot.resize(size);
217 return;
218 }
219 }
220
221 size=(int)mObjectsWorld.size();
222 for (int i=0; i<(int)mObjectsWorld.size(); ++i)
223 {
224 if (mObjectsWorld[i] && *mObjectsWorld[i]==name)
225 {
226 --size;
227 delete mObjectsWorld[i];
228 for (int j=i; j<size; ++j) mObjectsWorld[j]=mObjectsWorld[j+1];
229 mObjectsWorld.resize(size);
230 return;
231 }
232 }
233
234 size=(int)mTrajectoriesRoot.size();
235 for (int i=0; i<(int)mTrajectoriesRoot.size(); ++i)
236 {
237 if (mTrajectoriesRoot[i] && *mTrajectoriesRoot[i]==name)
238 {
239 --size;
240 delete mTrajectoriesRoot[i];
241 for (int j=i; j<size; ++j) mTrajectoriesRoot[j]=mTrajectoriesRoot[j+1];
242 mTrajectoriesRoot.resize(size);
243 return;
244 }
245 }
246
247 size=(int)mTrajectoriesWorld.size();
248 for (int i=0; i<(int)mTrajectoriesWorld.size(); ++i)
249 {
250 if (mTrajectoriesWorld[i] && *mTrajectoriesWorld[i]==name)
251 {
252 --size;
253 delete mTrajectoriesWorld[i];
254 for (int j=i; j<size; ++j) mTrajectoriesWorld[j]=mTrajectoriesWorld[j+1];
255 mTrajectoriesWorld.resize(size);
256 return;
257 }
258 }
259
260 return;
261 }
262
263 if (cmd=="object" || cmd=="object_with_label")
264 {
265 int idd=1;
266
267 std::string name(msg->get(idd++).asString().c_str());
268 std::string label(cmd=="object_with_label"?msg->get(idd++).asString().c_str():"");
269
270 double dx=msg->get(idd++).asFloat64();
271 double dy=msg->get(idd++).asFloat64();
272 double dz=msg->get(idd++).asFloat64();
273
274 double px=msg->get(idd++).asFloat64();
275 double py=msg->get(idd++).asFloat64();
276 double pz=msg->get(idd++).asFloat64();
277
278 double rx=msg->get(idd++).asFloat64();
279 double ry=msg->get(idd++).asFloat64();
280 double rz=msg->get(idd++).asFloat64();
281
282 int r=msg->get(idd++).asInt32();
283 int g=msg->get(idd++).asInt32();
284 int b=msg->get(idd++).asInt32();
285
286 double alpha=msg->get(idd++).asFloat64();
287
288 for (int i=0; i<(int)mObjectsRoot.size(); ++i)
289 {
290 if (mObjectsRoot[i] && *mObjectsRoot[i]==name)
291 {
292 mObjectsRoot[i]->set(dx,dy,dz,px,py,pz,rx,ry,rz,r,g,b,alpha,label);
293 return;
294 }
295 }
296
297 for (int i=0; i<(int)mObjectsWorld.size(); ++i)
298 {
299 if (mObjectsWorld[i] && *mObjectsWorld[i]==name)
300 {
301 mObjectsWorld[i]->set(dx,dy,dz,px,py,pz,rx,ry,rz,r,g,b,alpha,label);
302 return;
303 }
304 }
305
306 bool bWorld=(msg->size()>idd && msg->get(idd).asString()=="WORLD");
307
308 if (bWorld)
309 {
310 mObjectsWorld.push_back(new VisionObj(name,dx,dy,dz,px,py,pz,rx,ry,rz,r,g,b,alpha,label));
311 }
312 else
313 {
314 mObjectsRoot.push_back(new VisionObj(name,dx,dy,dz,px,py,pz,rx,ry,rz,r,g,b,alpha,label));
315 }
316
317 return;
318 }
319
320 if (cmd=="trajectory")
321 {
322 std::string name(msg->get(1).asString().c_str());
323 std::string label(msg->get(2).asString().c_str());
324
325 int bufflen=msg->get(3).asInt32();
326 double persistence=msg->get(4).asFloat64();
327
328 int R=msg->get(5).asInt32();
329 int G=msg->get(6).asInt32();
330 int B=msg->get(7).asInt32();
331
332 double alpha=msg->get(8).asFloat64();
333 GLfloat width=(GLfloat)msg->get(9).asFloat64();
334
335 for (int i=0; i<(int)mTrajectoriesRoot.size(); ++i)
336 {
337 if (mTrajectoriesRoot[i] && *mTrajectoriesRoot[i]==name)
338 {
339 mTrajectoriesRoot[i]->set(label,bufflen,persistence,R,G,B,alpha,width);
340 return;
341 }
342 }
343
344 for (int i=0; i<(int)mTrajectoriesWorld.size(); ++i)
345 {
346 if (mTrajectoriesWorld[i] && *mTrajectoriesWorld[i]==name)
347 {
348 mTrajectoriesWorld[i]->set(label,bufflen,persistence,R,G,B,alpha,width);
349 return;
350 }
351 }
352
353 bool bWorld=(msg->size()>10 && msg->get(10).asString()=="WORLD");
354
355 if (bWorld)
356 {
357 mTrajectoriesWorld.push_back(new TrajectoryObj(name,label,bufflen,persistence,R,G,B,alpha,width,bWorld));
358 }
359 else
360 {
361 mTrajectoriesRoot.push_back(new TrajectoryObj(name,label,bufflen,persistence,R,G,B,alpha,width,bWorld));
362 }
363
364 return;
365 }
366
367 if (cmd=="addpoint")
368 {
369 std::string name(msg->get(1).asString().c_str());
370
371 double x=msg->get(2).asFloat64();
372 double y=msg->get(3).asFloat64();
373 double z=msg->get(4).asFloat64();
374
375 for (int i=0; i<(int)mTrajectoriesRoot.size(); ++i)
376 {
377 if (mTrajectoriesRoot[i] && *mTrajectoriesRoot[i]==name)
378 {
379 mTrajectoriesRoot[i]->update(x,y,z);
380 return;
381 }
382 }
383
384 for (int i=0; i<(int)mTrajectoriesWorld.size(); ++i)
385 {
386 if (mTrajectoriesWorld[i] && *mTrajectoriesWorld[i]==name)
387 {
388 mTrajectoriesWorld[i]->update(x,y,z);
389 return;
390 }
391 }
392
393 return;
394 }

397void ObjectsManager::manage(iCub::skinDynLib::skinContactList &forces)
398{
399 for (int p=0; p<8; ++p)
400 {
401 for (int l=0; l<8; ++l)
402 {
403 if (mAB[p][l]) mAB[p][l]->clearArrows();
404 }
405 }
406
407 for (int i=0; i<(int)forces.size(); ++i)
408 {
409 yarp::sig::Vector P=forces[i].getCoP();
410
411 double f=forces[i].getForceModule();
412 yarp::sig::Vector F=forces[i].getForceDirection();
413 yarp::sig::Vector M=forces[i].getMoment();
414
415 int p=forces[i].getBodyPart();
416 int l=forces[i].getLinkNumber();
417
418 if (p >= 0 && p < 8 && l >= 0 && l < 8)
419 {
420 if (mAB[p][l])
421 {
422 mAB[p][l]->addArrow(new ForceArrow(P[0],P[1],P[2],f,F[0],F[1],F[2],M[0],M[1],M[2]));
423 }
424 }
425 else
426 {
427 yError("iCubGui: unexpected body part %d or link index %d contained in received skinContactList.", p, l);
428 }
429
430 }
431}
