383void iFTransformation::setTse()
384{
385 S=0.0;
386 Tse=0.0;
387 R=0.0;
388 for(int i=0; i<3;i++)
389 {
390 d(i)=(Hs(i,3)-He(i,3));
391 }
392 d=He.submatrix(0,2,0,2).transposed()*d;
393 S(0,1)=-d(2);
394 S(0,2)=d(1);
395 S(1,0)=d(2);
396 S(1,2)=-d(0);
397 S(2,0)=-d(1);
398 S(2,1)=d(0);
399
400 R=He.submatrix(0,2,0,2).transposed()*Hs.submatrix(0,2,0,2);
401 S=S*R;
402
403 for(int i=0; i<3;i++)
404 {
405 for(int j=0; j<3;j++)
406 {
407 Tse(i,j)=R(i,j);
408 Tse(i+3,j+3)=R(i,j);
409 Tse(i+3,j)=S(i,j);
410 }
411
412 }
