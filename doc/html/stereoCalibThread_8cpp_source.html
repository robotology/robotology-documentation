184bool StereoPairSynchronizer::tryPopPair(SynchronizedPair& pair)
185{
186 while(!leftQueue.empty() && !rightQueue.empty())
187 {
188 const double leftStamp = leftQueue.front().stamp.getTime();
189 const double rightStamp = rightQueue.front().stamp.getTime();
190
191 const double absoluteDelta = std::abs(leftStamp - rightStamp);
192 if(absoluteDelta <= toleranceSeconds)
193 {
194 pair.left = std::move(leftQueue.front().image);
195 pair.right = std::move(rightQueue.front().image);
196 pair.leftStamp = leftQueue.front().stamp;
197 pair.rightStamp = rightQueue.front().stamp;
198
199 pair.timeStampDelta = absoluteDelta;
200
201 leftQueue.pop_front();
202 rightQueue.pop_front();
203
204 std::lock_guard<std::mutex> lock(_mutex);
205 ++stats.pairedFrames;
206 stats.accumulatedTimeStampDelta += absoluteDelta;
207 stats.maxTimeStampDelta = std::max(stats.maxTimeStampDelta, absoluteDelta);
208
209 return true;
210 }
211
212 if(leftStamp < rightStamp)
213 {
214 // the oldest left frame cannot be paired with this or any newer right frame, so drop it
215 leftQueue.pop_front();
216 std::lock_guard<std::mutex> lock(_mutex);
217 ++stats.droppedLeftFrames;
218 }
219 else
220 {
221 rightQueue.pop_front();
222 std::lock_guard<std::mutex> lock(_mutex);
223 ++stats.droppedRightFrames;
224 }
225 }
226 return false;
227}

404void stereoCalibThread::processSynchronizedPair(SynchronizedPair& pair, Size boardSize)
405{
406 // Check minimum capture interval
407 const double pairTime = 0.5 * (pair.leftStamp.getTime() + pair.rightStamp.getTime()); //average pair timestamp
408 const double previousProcessedCandidateTime = lastProcessedCandidateTime;
409 if(previousProcessedCandidateTime >= 0.0 && (pairTime - previousProcessedCandidateTime) < minCaptureIntervalSeconds)
410 {
411 yCDebug(STEREOCALIBRATIONTHREAD) << "Skipping candidate pair due to minimum capture interval";
412 return;
413 }
414 if(previousProcessedCandidateTime >= 0.0)
415 {
416 yCDebug(STEREOCALIBRATIONTHREAD) << "Timestamp delta between processed pairs:" << (pairTime - previousProcessedCandidateTime) << "seconds";
417 }
418 lastProcessedCandidateTime = pairTime;
419
420
421 bool foundL=false;
422 bool foundR=false;
423 const Size leftSize(pair.left.width(), pair.left.height());
424 const Size rightSize(pair.right.width(), pair.right.height());
425
426 if(leftSize != _expectedImageSize || rightSize != _expectedImageSize)
427 {
428 if(_expectedImageSize.empty())
429 {
430 _expectedImageSize = leftSize;
431 }
432 }
433
434 if(leftSize != _expectedImageSize || rightSize != _expectedImageSize)
435 {
436 yCError(STEREOCALIBRATIONTHREAD) << "Left and right images have different sizes:" <<
437 "Left:" << leftSize.width << "x" << leftSize.height <<
438 "Right:" << rightSize.width << "x" << rightSize.height;
439 {
440 std::lock_guard<std::mutex> lock(mtx);
441 _calibrationError = "Input images do not match the expected calibration resolution.";
442 _calibrationResults = stereo_calib::CalibrationResult{};
443 }
444 calibrationState.store(CalibrationState::Error);
445
446 return;
447 }
448
449 LeftRgb=yarp::cv::toCvMat(pair.left);
450 RightRgb=yarp::cv::toCvMat(pair.right);
451
452 // Color adjust
453 Mat leftGray, rightGray;
454 cvtColor(LeftRgb,leftGray,CV_RGB2GRAY);
455 cvtColor(RightRgb,rightGray,CV_RGB2GRAY);
456
457 std::vector<Point2f> leftCorners;
458 std::vector<Point2f> rightCorners;
459
460 //TODO: for now we are only using CHESSBOARD GRID. For circle grid and so on we will add an update in the future.
461 if(boardType == "CIRCLES_GRID") {
462 foundL = findCirclesGrid(LeftRgb, boardSize, leftCorners, CALIB_CB_SYMMETRIC_GRID | CALIB_CB_CLUSTERING);
463 foundR = findCirclesGrid(RightRgb, boardSize, rightCorners, CALIB_CB_SYMMETRIC_GRID | CALIB_CB_CLUSTERING);
464 yCWarning(STEREOCALIBRATIONTHREAD) << "Board type:" << boardType << "not yet implemented.";
465 } else if(boardType == "ASYMMETRIC_CIRCLES_GRID") {
466 foundL = findCirclesGrid(LeftRgb, boardSize, leftCorners, CALIB_CB_ASYMMETRIC_GRID | CALIB_CB_CLUSTERING);
467 foundR = findCirclesGrid(RightRgb, boardSize, rightCorners, CALIB_CB_ASYMMETRIC_GRID | CALIB_CB_CLUSTERING);
468 yCWarning(STEREOCALIBRATIONTHREAD) << "Board type:" << boardType << "not yet implemented.";
469 } else if(boardType == "CHESSBOARD_SECTOR_BASED") {
470 foundL = findChessboardCornersSB(leftGray, boardSize, leftCorners);
471 foundR = findChessboardCornersSB(rightGray, boardSize, rightCorners);
472 yCWarning(STEREOCALIBRATIONTHREAD) << "Board type:" << boardType << "not yet implemented.";
473 } else {
474 foundL = findChessboardCorners(leftGray, boardSize, leftCorners, CALIB_CB_ADAPTIVE_THRESH | CALIB_CB_NORMALIZE_IMAGE | CALIB_CB_FILTER_QUADS);
475 foundR = findChessboardCorners(rightGray, boardSize, rightCorners, CALIB_CB_ADAPTIVE_THRESH | CALIB_CB_NORMALIZE_IMAGE | CALIB_CB_FILTER_QUADS);
476 }
477
478 if(foundL && foundR)
479 {
480 yCDebug(STEREOCALIBRATIONTHREAD) << "Found calibration board corners in both left and right images";
481
482 const Rect leftBoardBounds = boundingRect(leftCorners);
483 const Rect rightBoardBounds = boundingRect(rightCorners);
484 const bool leftBoardTooSmall =
485 leftBoardBounds.width < leftSize.width * minimumBoardSpanRatio ||
486 leftBoardBounds.height < leftSize.height * minimumBoardSpanRatio;
487 const bool rightBoardTooSmall =
488 rightBoardBounds.width < rightSize.width * minimumBoardSpanRatio ||
489 rightBoardBounds.height < rightSize.height * minimumBoardSpanRatio;
490
491 if(leftBoardTooSmall || rightBoardTooSmall)
492 {
493 yCWarning(STEREOCALIBRATIONTHREAD) << "Skipping stereo pair: chessboard is too small."
494 << "Left board:" << leftBoardBounds.width << "x" << leftBoardBounds.height
495 << "of" << leftSize.width << "x" << leftSize.height << ";"
496 << "right board:" << rightBoardBounds.width << "x" << rightBoardBounds.height
497 << "of" << rightSize.width << "x" << rightSize.height << "."
498 << "Each board must span at least" << (minimumBoardSpanRatio * 100.0)
499 << "% of both image dimensions.";
500
501 ++_rejectedDetections;
502 return;
503 }
504
505 TermCriteria criteria = TermCriteria(TermCriteria::EPS+TermCriteria::COUNT, 30, 0.01);
506 cornerSubPix(leftGray, leftCorners, Size(5,5), Size(-1,-1), criteria);
507 cornerSubPix(rightGray, rightCorners, Size(5,5), Size(-1,-1), criteria);
508
509 // Detection and geometry have succeeded. Build and validate the
510 // complete observation before assigning its accepted index.
511 stereo_calib::StereoObservation observation;
512 observation.imageSize = Size(pair.left.width(), pair.left.height());
513
514 observation.objectPoints = _chessboardConfiguration.createObjectPoints();
515 observation.leftImagePoints = leftCorners;
516 observation.rightImagePoints = rightCorners;
517
518 observation.leftTimestampSeconds = pair.leftStamp.getTime();
519 observation.rightTimestampSeconds = pair.rightStamp.getTime();
520 observation.timestampDeltaSeconds = pair.timeStampDelta;
521
522 observation.leftSequenceNumber = pair.leftStamp.getCount();
523 observation.rightSequenceNumber = pair.rightStamp.getCount();
524
525 if(!observation.isValid())
526 {
527 yCError(STEREOCALIBRATIONTHREAD) << "Generated invalid stereo observation";
528 ++_rejectedDetections;
529 return;
530 }
531
532 std::size_t observationIndex = 0;
533 observationIndex = _observations.size();
534
535 // Raw images are saved before drawing any optional diagnostics and
536 // only after the pair has become an accepted observation candidate.
537 if(_saveImages)
538 {
539 std::string imageError;
540 if(!_calibrationWriter.writeImagePair(imageDir, observationIndex,
541 LeftRgb, RightRgb,
542 observation.leftImageFilename,
543 observation.rightImageFilename,
544 imageError))
545 {
546 yCError(STEREOCALIBRATIONTHREAD) << "Could not save accepted calibration image pair:" << imageError;
547 {
548 std::lock_guard<std::mutex> lock(mtx);
549 if (calibrationState.load() != CalibrationState::Collecting)
550 {
551 return;
552 }
553
554 _calibrationError = imageError;
555 _calibrationResults = stereo_calib::CalibrationResult{};
556 calibrationState.store(CalibrationState::Error);
557 }
558 return;
559 }
560 }
561
562 // Detection and validation have completed.
563 {
564 std::lock_guard<std::mutex> lock(mtx);
565 if (calibrationState.load() != CalibrationState::Collecting)
566 {
567 return;
568 }
569 _observations.push_back(std::move(observation));
570 }
571
572 // Diagnostic overlays are deliberately the final step: they never
573 // affect the persisted raw dataset or the stored corner coordinates.
574 if(_drawDiagnosticCorners)
575 {
576 drawChessboardCorners(LeftRgb, boardSize, leftCorners, foundL);
577 drawChessboardCorners(RightRgb, boardSize, rightCorners, foundR);
578 }
579
580 ImageOf<PixelRgb>& outimL = outPortLeft.prepare();
581 outimL = pair.left;
582 outPortLeft.setEnvelope(pair.leftStamp);
583 outPortLeft.write();
584
585 ImageOf<PixelRgb>& outimR = outPortRight.prepare();
586 outimR = pair.right;
587 outPortRight.setEnvelope(pair.rightStamp);
588 outPortRight.write();
589
590 }
591 else
592 {
593 ++_rejectedDetections;
594 }
595
596 return;
597}

600{
601 StereoCalibStatus result;
602
603 std::lock_guard<std::mutex> lock(mtx);
604
605 switch(calibrationState.load())
606 {
607 case CalibrationState::Idle:
608 result.state = "Idle";
609 break;
610 case CalibrationState::Collecting:
611 result.state = "Collecting";
612 break;
613 case CalibrationState::Calibrating:
614 result.state = "Calibrating";
615 break;
616 case CalibrationState::Completed:
617 result.state = "Completed";
618 break;
619 case CalibrationState::Error:
620 result.state = "Error";
621 break;
622 }
623
624 const auto syncStats = synchronizer.getStatistics();
625
626 result.pairedFrames = syncStats.pairedFrames;
627 result.droppedLeftFrames = syncStats.droppedLeftFrames;
628 result.droppedRightFrames = syncStats.droppedRightFrames;
629
630 if(syncStats.pairedFrames > 0)
631 {
632 result.meanTimestampDeltaMs = 1000.0 * syncStats.accumulatedTimeStampDelta / static_cast<double>(syncStats.pairedFrames);
633 }
634
635 result.maxTimestampDeltaMs = 1000.0 * syncStats.maxTimeStampDelta;
636
637 result.lastCalibrationError = _calibrationError;
638 if(calibrationState.load() == CalibrationState::Completed && _calibrationResults.isValid())
639 {
640 result.calibrationAvailable = true;
641 switch(_calibrationResults.mode)
642 {
643 case stereo_calib::CalibrationMode::MonocularLeft:
644 result.calibrationMode = "MonocularLeft";
645 break;
646 case stereo_calib::CalibrationMode::MonocularRight:
647 result.calibrationMode = "MonocularRight";
648 break;
649 case stereo_calib::CalibrationMode::MonocularBoth:
650 result.calibrationMode = "MonocularBoth";
651 break;
652 case stereo_calib::CalibrationMode::StereoFull:
653 result.calibrationMode = "StereoFull";
654 break;
655 }
656
657 if(_calibrationResults.leftCamera.isValid())
658 {
659 result.leftMonocularRms = _calibrationResults.leftCamera.rms;
660 }
661 if(_calibrationResults.rightCamera.isValid())
662 {
663 result.rightMonocularRms = _calibrationResults.rightCamera.rms;
664 }
665 if(_calibrationResults.stereo.isValid())
666 {
667 result.stereoRms = _calibrationResults.stereo.rms;
668 result.baselineNorm = _calibrationResults.quality.baseline;
669 }
670 }
671
672 return result;

681void stereoCalibThread::stereoCalibRun()
682{
683 Size boardSize;
684 boardSize.width=this->boardWidth;
685 boardSize.height=this->boardHeight;
686
687 while (!isStopping())
688 {
689 // Keep the worker alive for RPC status inspection after a calibration
690 // failure, but stop consuming/processing camera data until a new start
691 // request changes the state back to Collecting.
692 if(calibrationState.load() == CalibrationState::Error)
693 {
694 Time::delay(0.1);
695 continue;
696 }
697
698 if(collectionResetRequested.exchange(false))
699 {
700 synchronizer.reset();
701 lastProcessedCandidateTime = -1.0;
702 _expectedImageSize = {};
703 _observations.clear();
704 _rejectedDetections = 0;
705 {
706 std::lock_guard<std::mutex> lock(mtx);
707 _calibrationResults = stereo_calib::CalibrationResult{};
708 _calibrationError.clear();
709 }
710 }
711
712 bool areFramesReceived = false;
713 ImageOf<PixelRgb> *tmpL = imagePortInLeft.read(false);
714
715 if(tmpL != nullptr)
716 {
717 areFramesReceived = true;
718
719 Stamp TSLeft;
720 imagePortInLeft.getEnvelope(TSLeft);
721
722 // Always publish preview immediately
723 ImageOf<PixelRgb>& outimL = outPortLeft.prepare();
724 outimL = *tmpL;
725 outPortLeft.setEnvelope(TSLeft);
726 outPortLeft.write();
727
728 // Only enqueue a separate copy while collecting
729 if(calibrationState.load() == CalibrationState::Collecting &&
730 shouldQueueFrameForCollection(TSLeft))
731 {
732 synchronizer.pushLeft(*tmpL, TSLeft); // this is done for not consuming memory bandwidth for data that will never be used by the calibration
733 }
734 }
735
736 ImageOf<PixelRgb> *tmpR = imagePortInRight.read(false);
737 if(tmpR!=nullptr)
738 {
739 areFramesReceived = true;
740
741 Stamp TSRight;
742 imagePortInRight.getEnvelope(TSRight);
743
744 // Always publish preview immediately
745 ImageOf<PixelRgb>& outimR=outPortRight.prepare();
746 outimR = *tmpR;
747 outPortRight.setEnvelope(TSRight);
748 outPortRight.write();
749
750
751 // Only enqueue a separate copy while collecting
752 if(calibrationState.load() == CalibrationState::Collecting &&
753 shouldQueueFrameForCollection(TSRight))
754 {
755 synchronizer.pushRight(*tmpR, TSRight);
756 }
757 }
758
759 std::vector<stereo_calib::StereoObservation> observationSnapshot;
760
761 if(calibrationState.load() == CalibrationState::Collecting)
762 {
763 SynchronizedPair pair;
764 while(calibrationState.load() == CalibrationState::Collecting &&
765 synchronizer.tryPopPair(pair))
766 {
767 // Process the synchronized pair
768 processSynchronizedPair(pair, boardSize);
769
770 if(calibrationState.load() != CalibrationState::Collecting)
771 {
772 break;
773 }
774 if(_observations.size() >= static_cast<std::size_t>(numOfPairs))
775 {
776 yCInfo(STEREOCALIBRATIONTHREAD, "Collected %zu valid stereo observations. Stopping collection.", _observations.size());
777 observationSnapshot = _observations;
778 calibrationState.store(CalibrationState::Calibrating);
779 yCInfo(STEREOCALIBRATIONTHREAD) << "Observation collection complete";
780 break;
781 }
782 }
783 }
784
785 if(calibrationState.load() == CalibrationState::Calibrating)
786 {
787 yCInfo(STEREOCALIBRATIONTHREAD) << "Starting the calibration process";
788 if(observationSnapshot.empty())
789 {
790 observationSnapshot = _observations;
791 }
792
793 stereo_calib::CalibrationResult calibrationResult;
794 std::string calibrationError;
795 bool success = false;
796 if (_cameraModel == stereo_calib::CameraModel::Pinhole)
797 {
798 stereo_calib::PinholeCalibrationOptions options = _pinholeCalibrationOptions;
799 options.common.calibrationMode = _calibrationMode;
800 options.common.imageSize = observationSnapshot.front().imageSize;
801 success = _pinholeCalibrationEngine.calibrate(
802 observationSnapshot,
803 options,
804 calibrationResult,
805 calibrationError
806 );
807 }
808 else if(_cameraModel == stereo_calib::CameraModel::Fisheye)
809 {
810 stereo_calib::FisheyeCalibrationOptions options = _fisheyeCalibrationOptions;
811 options.common.calibrationMode = _calibrationMode;
812 options.common.imageSize = observationSnapshot.front().imageSize;
813 options.cameraFocalLengthGuess = 625.0;
814 success = _fisheyeCalibrationEngine.calibrate(
815 observationSnapshot,
816 options,
817 calibrationResult,
818 calibrationError
819 );
820 }
821 else
822 {
823 calibrationError = "Unsupported camera model.";
824 success = false;
825 }
826
827 // success and writer should be common between fisheye and pinhole engines, so this block can be shared
828 if(!success)
829 {
830 // The engine converts OpenCV exceptions into a diagnostic. Log
831 // it here, where YARP logging is allowed, and stop calibration
832 // processing while keeping the Error state and status available.
833 yCError(STEREOCALIBRATIONTHREAD) << "Calibration failed:" << calibrationError;
834 {
835 std::lock_guard<std::mutex> lock(mtx);
836 _calibrationResults = stereo_calib::CalibrationResult{};
837 _calibrationError = calibrationError.empty()
838 ? "Calibration failed without an error message."
839 : calibrationError;
840 }
841 calibrationState.store(CalibrationState::Error);
842 continue;
843 }
844
845 const auto syncStats = synchronizer.getStatistics();
846 calibrationResult.quality.rejectedDetections = _rejectedDetections;
847 calibrationResult.quality.acceptedObservations = observationSnapshot.size();
848 calibrationResult.quality.synchronizedPairs = syncStats.pairedFrames;
849 if(syncStats.pairedFrames > 0)
850 {
851 calibrationResult.quality.meanTimestampDeltaMs = 1000.0 * syncStats.accumulatedTimeStampDelta / static_cast<double>(syncStats.pairedFrames);
852 }
853 calibrationResult.quality.maxTimestampDeltaMs = syncStats.maxTimeStampDelta;
854 calibrationResult.quality.baseline = cv::norm(calibrationResult.stereo.T);
855
856 std::string persistenceError;
857 if(!_calibrationWriter.write(camCalibFile, calibrationResult, persistenceError))
858 {
859 yCError(STEREOCALIBRATIONTHREAD) << "Could not save calibration results:" << persistenceError;
860 {
861 std::lock_guard<std::mutex> lock(mtx);
862 _calibrationResults = std::move(calibrationResult);
863 _calibrationError = persistenceError;
864 }
865 calibrationState.store(CalibrationState::Error);
866 continue;
867 }
868 if(!_calibrationWriter.writeObservations(_observationsFile, observationSnapshot, persistenceError))
869 {
870 yCError(STEREOCALIBRATIONTHREAD) << "Could not save calibration observations:" << persistenceError;
871 {
872 std::lock_guard<std::mutex> lock(mtx);
873 _calibrationResults = std::move(calibrationResult);
874 _calibrationError = persistenceError;
875 }
876 calibrationState.store(CalibrationState::Error);
877 continue;
878 }
879
880 logCalibrationResult(calibrationResult);
881 {
882 std::lock_guard<std::mutex> lock(mtx);
883 _calibrationResults = std::move(calibrationResult);
884 _calibrationError.clear();
885 }
886
887 calibrationState.store(CalibrationState::Completed);
888 yCInfo(STEREOCALIBRATIONTHREAD) << "Entire calibration process completed";
889 }
890
891 if(!areFramesReceived)
892 {
893 Time::delay(0.001); // Sleep for a short duration to avoid busy waiting
894 }
895 cout.flush();
896 }
897}
