124private:
125 uint32_t _threshold; // threshold for the delta between current and previous temperature --> set to 20 Celsius deg by default --> over 20 deg delta spike
126 double _motorTempPrev; // motor temperature at previous instant for checking positive temperature spikes
127 bool _isStarted;
128 int32_t _initCounter;
129 std::vector<double> _initTempBuffer;
130public:
131 TemperatureFilter(): _threshold(20), _isStarted(false), _initCounter(50), _initTempBuffer(0), _motorTempPrev(0){;}
132 TemperatureFilter(uint32_t threshold, int32_t initCounter): _threshold(threshold), _isStarted(false), _initCounter(initCounter), _initTempBuffer(0), _motorTempPrev(0){;}
133 ~TemperatureFilter() = default;
134 TemperatureFilter(const TemperatureFilter& other) = default;
135 TemperatureFilter(TemperatureFilter&& other) noexcept = default;
136 TemperatureFilter& operator=(const TemperatureFilter& other) = default;
137 TemperatureFilter& operator=(TemperatureFilter&& other) noexcept = default;
138
139 bool isStarted(){return _isStarted;}
140 uint32_t getTemperatureThreshold() {return _threshold; }
141 double getPrevTemperature(){return _motorTempPrev;}
142 void updatePrevTemperature(double temperature){_motorTempPrev = temperature;}
143 void start(double temperature)
144 {
145 if(_initCounter < 0)
146 {
147 int median_pos = std::ceil(_initTempBuffer.size() / 2) -1;
148 _motorTempPrev = _initTempBuffer.at(median_pos);
149 _isStarted = true;
150 }
151 else
152 {
153 _initTempBuffer.push_back(temperature);
154 --_initCounter;
155 }
156
157 }
