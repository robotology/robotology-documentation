53 enum {
54 EMBSK_SIZE_INFO = 128,

56 MAX_NUM_OF_PATCHES_PER_ETH_BOARD = 2, // this is the maximum number of patches that can be associated to a single eth board. This is a limit that we can set since it is not possible to have more than 2 patches per board (since each patch corresponds to a can port and each board has 2 can ports)
57 SKIN_TAXELS_PER_TRIANGLE = 12, // this is the number of taxels per triangle, it is a fixed value since it depends on the hardware design of the skin, and it is not expected to change in the future.
58 TRIANGLES_PER_CAN_BOARD = 16 // this is the number of triangles per can board, it is a fixed value since it depends on the hardware design of the skin, and it is not expected to change in the future. Each can board has 16 triangles, each triangle has 12 taxels, so each can board has 192 taxels.
59 };

115bool EmbObjSkin::initWithSpecialConfig(yarp::os::Searchable& config)
116{
117 eOprotID32_t protoid;
118 std::vector<SpecialSkinBoardCfgParam> boardCfgList(_skCfg.totalCardsNum);
119 unsigned int numofcfg;
120 unsigned int j = 0, p = 0;
121
122 if(!_newCfg)
123 {
124 return true; //if we use old style config then return
125 }
126
127 //-----------------------------------------------------------------------------------------------------
128 //------------ read special cfg board --------------------------------------------------------------
129
130 numofcfg = _skCfg.totalCardsNum;//set size of my vector boardCfgList;
131 //in output the function return number of special board cfg are in file xml
132 // TODO: add documentation regarding usage of special configuration in xml file
133 bool ret = _cfgReader.readSpecialBoardCfg(config, boardCfgList.data(), &numofcfg);
134
135 if(!ret)
136 return false;
137
138
139 for(j=0; j<numofcfg; j++) //for each special board config
140 {
141 //check if patch exist
142 for(p=0; p< _skCfg.patchInfoList.size(); p++)
143 {
144 if(_skCfg.patchInfoList[p].idPatch == boardCfgList[j].patch)
145 break;
146 }
147 if(p>=_skCfg.patchInfoList.size())
148 {
149 yCError(EMBOBJSKIN) << __YFUNCTION__ << ":" << boardInfo() << ": patch" << boardCfgList[j].patch << "not exists";
150 return false;
151 }
152 //now p is the index of patch.
153
154 eOcanport_t canport = _skCfg.patchInfoList[p].canport;
155
156 //check if card address are in patch
157 std::optional<size_t> boardIdx;
158 for(int a=boardCfgList[j].boardAddrStart; a<=boardCfgList[j].boardAddrEnd; a++)
159 {
160 boardIdx = _skCfg.patchInfoList[p].checkCardAddrIsInList(a);
161 if(!boardIdx)
162 {
163 yCError(EMBOBJSKIN) << __YFUNCTION__ << ":" << boardInfo() << "card with address" << a << "is not present in patch" << _skCfg.patchInfoList[p].idPatch;
164 return(false);
165 }
166 }
167
168 // prepare data to send to ems
169 eOsk_cmd_boardsCfg_t bcfg;
170 // this message is for some addresses only
171 bcfg.candestination[0] = bcfg.candestination[1] = 0;
172 for(int adr=boardCfgList[j].boardAddrStart; adr<=boardCfgList[j].boardAddrEnd; adr++)
173 {
174 if(isCANaddressValid(adr))
175 {
176 eo_common_hlfword_bitset(&bcfg.candestination[canport], adr);
177 }
178 }
179
180 bcfg.cfg.skintype = boardCfgList[j].cfg.skinType;
181 bcfg.cfg.period = boardCfgList[j].cfg.period;
182 bcfg.cfg.noload = boardCfgList[j].cfg.noLoad;
183
184 // Init the data vector with special config values from "noLoad" param in config file.
185 // This is to have a correct initilization for the data sent through yarp port
186 for (size_t sensorId = 0; sensorId < TRIANGLES_PER_CAN_BOARD; sensorId++)
187 {
188 size_t index = TRIANGLES_PER_CAN_BOARD * SKIN_TAXELS_PER_TRIANGLE * (*boardIdx) + sensorId * SKIN_TAXELS_PER_TRIANGLE;
189 yCTrace(EMBOBJSKIN) << __YFUNCTION__ << ":" << boardInfo() << ": index is" << index << "value is:" << boardCfgList[j].cfg.noLoad;
190
191 // Message head
192 for(int k = 0; k < SKIN_TAXELS_PER_TRIANGLE; k++)
193 {
194 yCTrace(EMBOBJSKIN) << __YFUNCTION__ << ":" << boardInfo() << ": size is:" << skindata.size() << "index is" << (index+k) << "value is:" << boardCfgList[j].cfg.noLoad;
195 if((index+k) >= skindata.size())
196 {
197 yCError(EMBOBJSKIN) << __YFUNCTION__ << ":" << boardInfo() << ": index too big";
198 }
199 skindata[index + k] = boardCfgList[j].cfg.noLoad;
200 }
201 }
202
203 protoid = eoprot_ID_get(eoprot_endpoint_skin, eoprot_entity_sk_skin, _skCfg.patchInfoList[p].indexNv, eoprot_tag_sk_skin_cmmnds_boardscfg);
204
205 if(false == res->setRemoteValue(protoid, &bcfg))
206 {
207 yCError(EMBOBJSKIN) << __YFUNCTION__ << ":" << boardInfo() << "Error in send special board config for mtb with addr from" << boardCfgList[j].boardAddrStart << "to addr" << boardCfgList[j].boardAddrEnd;
208 return false;
209 }
210
211 } //end for for each special board cfg
212
213 SystemClock::delaySystem(0.010); // 10 ms (m.a.a-delay: before it was 0.01)
214
215 //-----------------------------------------------------------------------------------------------------
216 //------------ read special cfg triangle --------------------------------------------------------------
217 std::vector<SpecialSkinTriangleCfgParam> triangleCfg(SPECIAL_TRIANGLE_CFG_MAX_NUM);
218 numofcfg = SPECIAL_TRIANGLE_CFG_MAX_NUM; //set size of my vector triangleCfg;
219 //in output the function return number of special triangle cfg are in file xml
220 ret = _cfgReader.readSpecialTriangleCfg(config, triangleCfg.data(), &numofcfg);
221 if(!ret)
222 return false;
223
224
225 for(j=0; j<numofcfg; j++)
226 {
227 for(p=0; p< _skCfg.patchInfoList.size(); p++)
228 {
229 if(_skCfg.patchInfoList[p].idPatch == triangleCfg[j].patch)
230 break;
231 }
232 if(p >= _skCfg.patchInfoList.size())
233 {
234 yCError(EMBOBJSKIN) << __YFUNCTION__ << ":" << boardInfo() << ":patch" << triangleCfg[j].patch << "not exists";
235 return false;
236 }
237 //now p is index patch
238
239 eOcanport_t canport = _skCfg.patchInfoList[p].canport;
240
241 //check if bcfg.boardAddr is in my patches list
242 if(!_skCfg.patchInfoList[p].checkCardAddrIsInList(triangleCfg[j].boardAddr))
243 {
244 yCError(EMBOBJSKIN) << __YFUNCTION__ << ":" << boardInfo() << "card with address" << triangleCfg[j].boardAddr << "is not present in patch" << _skCfg.patchInfoList[p].idPatch;
245 return(false);
246 }
247
248 //prepare data to send to ems
249 protoid = eoprot_ID_get(eoprot_endpoint_skin, eoprot_entity_sk_skin, _skCfg.patchInfoList[p].indexNv, eoprot_tag_sk_skin_cmmnds_trianglescfg);
250
251
252 eOsk_cmd_trianglesCfg_t tcfg = {0};
253
254 // this message is for one address only
255 tcfg.candestination[0] = tcfg.candestination[1] = 0;
256 int adr = triangleCfg[j].boardAddr;
257 if(isCANaddressValid(adr))
258 {
259 eo_common_hlfword_bitset(&tcfg.candestination[canport], adr);
260 }
261
262 tcfg.idstart = triangleCfg[j].triangleStart;
263 tcfg.idend = triangleCfg[j].triangleEnd;
264 tcfg.cfg.CDCoffset = triangleCfg[j].cfg.cdcOffset;
265 tcfg.cfg.enable = triangleCfg[j].cfg.enabled;
266 tcfg.cfg.shift = triangleCfg[j].cfg.shift;
267
268
269 SystemClock::delaySystem(0.010);
270
271 if(false == res->setRemoteValue(protoid, &tcfg))
272 {
273 yCError(EMBOBJSKIN) << __YFUNCTION__ << ":" << boardInfo() << "Error in send special triangle config for board CAN" << canport+1 << ":" << triangleCfg[j].boardAddr;
274 return false;
275 }
276 }
277
278 SystemClock::delaySystem(0.010); // 10 ms (m.a.a-delay: before it was 0.01)
279
280 return true;
281}

284bool EmbObjSkin::fromConfig(yarp::os::Searchable& config)
285{
286 Bottle bPatches, bPatchList, xtmp;
287 //reset total num of cards
288 // A card is the number of each entry skinCanAddrsPatchX inside the patches group --> i.e. the number of CAN addresses found in config file. We should change the name of this variable with something more meaningful.
289 _skCfg.totalCardsNum = 0;
290 _cumulativeTaxels = 0;
291
292
293 servConfigSkin_t skinconfig;
294 parser->parseService(config, skinconfig);
295
296 bPatches = config.findGroup("patches", "skin patches connected to this device");
297 if(bPatches.isNull())
298 {
299 yCError(EMBOBJSKIN) << __YFUNCTION__ << ":" << boardInfo() << "patches group is missing";
300 return(false);
301 }
302
303 _skCfg.numOfPatches=0;
304 for (int i=1; i<=MAX_NUM_OF_PATCHES_PER_ETH_BOARD; i++)
305 {
306 std::string skinCanAddrsPatchx = "skinCanAddrsPatch";
307 skinCanAddrsPatchx.append(std::to_string(i));
308 if (bPatches.check(skinCanAddrsPatchx))
309 {
310 _skCfg.numOfPatches++; //we increase the number of patches per each instance of skinCanAddrsPatchX found in config file.
311 bPatchList.addInt32(i);
312 }
313 }
314
315 _skCfg.patchInfoList.clear();
316 _skCfg.patchInfoList.resize(_skCfg.numOfPatches);
317 for(int j=1; j<_skCfg.numOfPatches+1; j++)
318 {
319 int id = bPatchList.get(j-1).asInt32();
320 if((id!=1) && (id!=2))
321 {
322 yCError(EMBOBJSKIN) << __YFUNCTION__ << ":" << boardInfo() << "expecting at most 2 patches";
323 return false;
324 }
325 _skCfg.patchInfoList[j-1].idPatch = id;
326 _skCfg.patchInfoList[j-1].indexNv = convertIdPatch2IndexNv(id);
327 _skCfg.patchInfoList[j-1].canport = (1 == id) ? eOcanport1 : eOcanport2;
328 }
329
330
331 for(int i=0; i<_skCfg.numOfPatches; i++)
332 {
333 char tmp[80];
334 int id = _skCfg.patchInfoList[i].idPatch;
335 snprintf(tmp, sizeof(tmp), "skinCanAddrsPatch%d", id);
336
337 xtmp = bPatches.findGroup(tmp);
338 if(xtmp.isNull())
339 {
340 yCError(EMBOBJSKIN) << __YFUNCTION__ << ":" << boardInfo() << "does not find " << tmp << "in xml file";
341 return false;
342 }
343
344 _skCfg.patchInfoList[i].cardAddrList.resize(xtmp.size()-1);
345 _skCfg.patchInfoList[i].taxelsSize = _skCfg.patchInfoList[i].cardAddrList.size() * TRIANGLES_PER_CAN_BOARD * SKIN_TAXELS_PER_TRIANGLE;
346 _skCfg.patchInfoList[i].taxelsOffset = _cumulativeTaxels;
347 _cumulativeTaxels += _skCfg.patchInfoList[i].taxelsSize;
348 for(int j=1; j<xtmp.size(); j++)
349 {
350 int addr = xtmp.get(j).asInt32();
351 _skCfg.totalCardsNum++;
352 _skCfg.patchInfoList[i].cardAddrList[j-1] = addr;
353 }
354 }
355 // impose the number of sensors (triangles found in config file)
356 sensorsNum = _cumulativeTaxels;
357 // sensorsNum is the total number of taxels/sensors on this ETH board.
358
359 // resize the skindata holder
360 mtx.lock();
361
362 this->skindata.resize(sensorsNum);
363 int ttt = this->skindata.size();
364 for (int i=0; i < ttt; i++)
365 {
366 this->skindata[i]=defaultSkinNoLoadValue; // skindata contains data for all the cards. 240 is the max value for a taxel, i.e. 0% pressure on it.
367 }
368
369 mtx.unlock();
370
371 // fill the ethservice ...
372
373 ethservice.configuration.type = eomn_serv_SK_skin;
374
375
376 ethservice.configuration.data.sk.skin.boardinfo.type = skinconfig.canboard.type;
377
378 ethservice.configuration.data.sk.skin.boardinfo.protocol.major = skinconfig.canboard.protocol.major;
379 ethservice.configuration.data.sk.skin.boardinfo.protocol.minor = skinconfig.canboard.protocol.minor;
380
381 ethservice.configuration.data.sk.skin.boardinfo.firmware.major = skinconfig.canboard.firmware.major;
382 ethservice.configuration.data.sk.skin.boardinfo.firmware.minor = skinconfig.canboard.firmware.minor;
383 ethservice.configuration.data.sk.skin.boardinfo.firmware.build = skinconfig.canboard.firmware.build;
384
385
386 ethservice.configuration.data.sk.skin.numofpatches = _skCfg.numOfPatches;
387 if(ethservice.configuration.data.sk.skin.numofpatches > eomn_serv_skin_maxpatches)
388 {
389 yCError(EMBOBJSKIN) << __YFUNCTION__ << ":" << boardInfo() << ": Cannot have so many skin patches. detected" << ethservice.configuration.data.sk.skin.numofpatches << "max is" << eomn_serv_skin_maxpatches;
390 return false;
391 }
392 // tagliato e cucito per le schede eb2 eb4 eb10 ed eb11. da migliorare sia i file xml che il parser.
393 for(int np=0; np<ethservice.configuration.data.sk.skin.numofpatches; np++)
394 {
395 // patch np-th, can1 ... for each address put a bit using eo_common_hlfword_bitset
396 ethservice.configuration.data.sk.skin.canmapskin[np][0] = 0;
397 //eo_common_hlfword_bitset(&ethservice.configuration.data.sk.skin.canmapskin[np][0], 3);
398
399 // patch np-th, can2 ... for each address put a bit using
400 ethservice.configuration.data.sk.skin.canmapskin[np][1] = 0;
401
402 eOcanport_t canport = _skCfg.patchInfoList[np].canport;
403
404 int max = _skCfg.patchInfoList[np].cardAddrList.size();
405 for(int n=0; n<max; n++)
406 {
407 int adr = _skCfg.patchInfoList[np].cardAddrList.at(n);
408 adr = adr;
409 if(isCANaddressValid(adr))
410 {
411 eo_common_hlfword_bitset(&ethservice.configuration.data.sk.skin.canmapskin[np][canport], adr);
412 }
413 }
414 }
415
416
417 if( _cfgReader.isDefaultBoardCfgPresent(config) && _cfgReader.isDefaultTriangleCfgPresent(config))
418 {
419 _newCfg = true;
420 }
421 else
422 {
423 _newCfg = false;
424 return true;
425 }
426
427 /*read skin board default configuration*/
428 _brdCfg.setDefaultValues();
429 if(!_cfgReader.readDefaultBoardCfg(config, &_brdCfg))
430 return false;
431
432 // Fill the data vector with default values from "noLoad" param in config file.
433 for (int board_idx = 0; board_idx < _skCfg.totalCardsNum; board_idx++)
434 {
435 for (int triangleId = 0; triangleId < TRIANGLES_PER_CAN_BOARD; triangleId++)
436 {
437 int index = TRIANGLES_PER_CAN_BOARD * SKIN_TAXELS_PER_TRIANGLE * board_idx + triangleId * SKIN_TAXELS_PER_TRIANGLE;
438
439 // Message head
440 for (size_t k = 0; k < SKIN_TAXELS_PER_TRIANGLE; k++)
441 {
442 if((index+k) >= skindata.size())
443 yCError(EMBOBJSKIN) << __YFUNCTION__ << ":" << boardInfo() << ": index too big";
444 skindata[index + k] = _brdCfg.noLoad;
445 }
446 }
447 }
448
449 /*read skin triangle default configuration*/
450 _triangCfg.setDefaultValues();
451 if(! _cfgReader.readDefaultTriangleCfg(config, &_triangCfg))
452 return false;
453
454 return true;
455}

458bool EmbObjSkin::open(yarp::os::Searchable& config)
459{
460 // - first thing to do is verify if the eth manager is available. then i parse info about the eth board.
461
462 ethManager = eth::TheEthManager::instance();
463 if(NULL == ethManager)
464 {
465 yCFatal(EMBOBJSKIN) << __YFUNCTION__ << ": fails to instantiate ethManager";
466 return false;
467 }
468
469 if(false == ethManager->verifyEthBoardInfo(config, ipv4addr, boardIPstring, boardName))
470 {
471 yCError(EMBOBJSKIN) << __YFUNCTION__ << ":" << boardInfo() << ": object TheEthManager fails in parsing ETH properties from xml file";
472 return false;
473 }
474 // add specific info about this device ...
475
476 // - now all other things
477
478 if(NULL == parser)
479 {
480 parser = new ServiceParser;
481 }
482
483 // read config file
484 if(false == fromConfig(config))
485 {
486 yCError(EMBOBJSKIN) << __YFUNCTION__ << ": fails in fromConfig():" << boardInfo() << ": cannot proceed any further";
487 cleanup();
488 return false;
489 }
490
491
492
493 // -- instantiate EthResource etc.
494
495 res = ethManager->requestResource2(this, config);
496 if(NULL == res)
497 {
498 yCError(EMBOBJSKIN) << __YFUNCTION__ << ":" << boardInfo() << ": fails because could not instantiate the ethResource for BOARD w/ IP =" << boardIPstring << "... unable to continue";
499 return false;
500 }
501
502 // now we have an ip address, thus we can set the name in object SkinConfigReader
503 char name[80];
504 snprintf(name, sizeof(name), "embObjSkin on BOARD %s IP %s", res->getProperties().boardnameString.c_str(), res->getProperties().ipv4addrString.c_str());
505 _cfgReader.setName(name);
506
507
508 if(!res->verifyEPprotocol(eoprot_endpoint_skin))
509 {
510 cleanup();
511 return false;
512 }
513
514
515 const eOmn_serv_parameter_t* servparam = &ethservice;
516
517 if(false == res->serviceVerifyActivate(eomn_serv_category_skin, servparam, 5.0))
518 {
519 yCError(EMBOBJSKIN) << __YFUNCTION__ << ": error in ethResources::serviceVerifyActivate() for" << boardInfo();
520 cleanup();
521 return false;
522 }
523
524 if(!init())
525 return false;
526
527 /* Following delay is necessary in order to give enough time to skin boards to configure all its triangles */
528 SystemClock::delaySystem(0.500);
529
530 if(!initWithSpecialConfig(config))
531 {
532 cleanup();
533 return false;
534 }
535
536 if(!configPeriodicMessage())
537 {
538 cleanup();
539 return false;
540 }
541
542 if(!start())
543 {
544 cleanup();
545 return false;
546 }
547
548 if(false == res->serviceStart(eomn_serv_category_skin))
549 {
550 yCError(EMBOBJSKIN) << __YFUNCTION__ << ": fails to start skin service for" << boardInfo() << ": cannot continue";
551 cleanup();
552 return false;
553 }
554 else
555 {
556 if(verbosewhenok)
557 {
558 yCDebug(EMBOBJSKIN) << __YFUNCTION__ << ": correctly starts skin service of" << boardInfo();
559 }
560 }
561
562 return true;
563}

805bool EmbObjSkin::update(eOprotID32_t id32, double timestamp, void *rxdata)
806{
807 uint8_t msgtype = 0;
808 uint8_t i, triangle = 0;
809 static int error = 0;
810 int p;
811 EOarray* arrayof = (EOarray*)rxdata;
812 uint8_t sizeofarray = eo_array_Size(arrayof);
813
814 eOprotIndex_t indexpatch = eoprot_ID2index(id32);
815
816 for(p=0; p<_skCfg.numOfPatches; p++)
817 {
818 if(_skCfg.patchInfoList[p].indexNv == indexpatch)
819 break;
820 }
821 if(p >= _skCfg.numOfPatches)
822 {
823 yCError(EMBOBJSKIN) << __YFUNCTION__ << ":" << boardInfo() << ": received data of patch with nvindex=" << indexpatch;
824 return false;
825 }
826
827 errors.resize(sizeofarray);
828
829 for(i=0; i<sizeofarray; i++)
830 {
831 eOsk_candata_t *candata = (eOsk_candata_t*) eo_array_At(arrayof, i);
832
833 if(NULL == candata)
834 {
835 break;
836 }
837
838 uint16_t canframeid11 = EOSK_CANDATA_INFO2IDCAN(candata->info);
839 uint8_t canframesize = EOSK_CANDATA_INFO2SIZE(candata->info);
840 uint8_t *canframedata = candata->data;
841
842 uint8_t cardAddr = 0;
843 uint8_t valid = 0;
844 uint8_t skinClass;
845
846 if(_newCfg)
847 skinClass = ICUBCANPROTO_CLASS_PERIODIC_SKIN;
848 else
849 skinClass = ICUBCANPROTO_CLASS_PERIODIC_ANALOGSENSOR;
850
851
852
853 valid = (((canframeid11 & 0x0f00) >> 8) == skinClass) ? 1 : 0;
854
855 if(valid)
856 {
857 cardAddr = (canframeid11 & 0x00f0) >> 4;
858 size_t localBoardIndex = 0;
859 bool foundCardAddr = false;
860 for (size_t cId_index = 0; cId_index< _skCfg.patchInfoList[p].cardAddrList.size(); cId_index++)
861 {
862 if(_skCfg.patchInfoList[p].cardAddrList[cId_index] == cardAddr)
863 {
864 localBoardIndex = cId_index;
865 foundCardAddr = true;
866 break;
867 }
868 }
869
870 if(!foundCardAddr)
871 {
872 yCError(EMBOBJSKIN) << __YFUNCTION__ << ":" << boardInfo() << ": Unknown cardId from skin";
873 return false;
874 }
875
876 triangle = (canframeid11 & 0x000f);
877 msgtype = (int) canframedata[0];
878
879 const size_t patchBase = _skCfg.patchInfoList[p].taxelsOffset;
880 const size_t boardBase = localBoardIndex * TRIANGLES_PER_CAN_BOARD * SKIN_TAXELS_PER_TRIANGLE;
881 const size_t index = patchBase + boardBase + triangle * SKIN_TAXELS_PER_TRIANGLE;
882
883 // marco.accame: added lock to avoid concurrent access to this->skindata. i lock at triangle resolution ...
884 mtx.lock();
885
886 if (msgtype == 0x40)
887 {
888#if defined(DEBUG_PRINT_RX_STATS)
889 receivedpatches[p][cardAddr]++;
890 counterpa ++;
891#endif
892 // Message head
893 for(int k = 0; k < 7; k++)
894 {
895 skindata[index + k] = canframedata[k + 1];
896 }
897 }
898 else if (msgtype == 0xC0)
899 {
900 // Message tail
901 for(int k = 0; k < 5; k++)
902 {
903 skindata[index + k + 7] = canframedata[k + 1];
904 }
905
906 // Skin diagnostics
907 if (_brdCfg.useDiagnostic) // if user requests to check the diagnostic
908 {
909 if (canframesize == 8)
910 {
911 // Skin diagnostics is active
912 _isDiagnosticPresent = true;
913
914 // Get error code head and tail
915 short head = canframedata[6];
916 short tail = canframedata[7];
917 int fullMsg = (head << 8) | (tail & 0xFF);
918
919 // Store error message
920 errors[i].net = indexpatch;
921 errors[i].board = cardAddr;
922 errors[i].sensor = triangle;
923 errors[i].error = fullMsg;
924
925 if (fullMsg != SkinErrorCode::StatusOK)
926 {
927 yCError(EMBOBJSKIN) << __YFUNCTION__ << ":" << boardInfo() <<
928 "canDeviceNum: " << errors[i].net <<
929 "board: " << errors[i].board <<
930 "sensor: " << errors[i].sensor <<
931 "error: " << iCub::skin::diagnostics::printErrorCode(errors[i].error).c_str();
932 }
933 }
934 else
935 {
936 _isDiagnosticPresent = false;
937 }
938 }
939 }
940 mtx.unlock();
941 }
942 else if(canframeid11 == 0x100)
943 {
944 /* Can frame with id =0x100 contains Debug info. SO I skip it.*/
945 return true;
946 }
947 else
948 {
949 if(error == 0)
950 yCError(EMBOBJSKIN) << __YFUNCTION__ << ":" << boardInfo() << "Unknown Message received from skin (" << i << "/" << sizeofarray << "): frameID=" << canframeid11 << " len=" << canframesize << "canframe.data=" << canframedata[0] << " " << canframedata[1] << " " << canframedata[2] << " " << canframedata[3];
951 error++;
952 if (error == 10000)
953 error = 0;
954 }
955 }
956
957#if defined(DEBUG_PRINT_RX_STATS)
958 if(counterpa >= 10000)
959 {
960 counterpa = 0;
961 yDebug("pa = {{%d %d %d %d %d %d %d %d %d %d %d %d %d %d %d %d} {%d %d %d %d %d %d %d %d %d %d %d %d %d %d %d %d}}",
962 receivedpatches[0][0], receivedpatches[0][1], receivedpatches[0][2], receivedpatches[0][3], receivedpatches[0][4], receivedpatches[0][5], receivedpatches[0][6], receivedpatches[0][7],
963 receivedpatches[0][8], receivedpatches[0][9], receivedpatches[0][10], receivedpatches[0][11], receivedpatches[0][12], receivedpatches[0][13], receivedpatches[0][14], receivedpatches[0][15],
964 receivedpatches[1][0], receivedpatches[1][1], receivedpatches[1][2], receivedpatches[1][3], receivedpatches[1][4], receivedpatches[1][5], receivedpatches[1][6], receivedpatches[1][7],
965 receivedpatches[1][8], receivedpatches[1][9], receivedpatches[1][10], receivedpatches[1][11], receivedpatches[1][12], receivedpatches[1][13], receivedpatches[1][14], receivedpatches[1][15]
966 );
967
968 }
969#endif
970
971 return true;
